-- Cumulative Sales Analysis
-- Calculate total sales for each year and track the cumulative sales over time.

SELECT
    YEAR(order_date) AS [Order Year],

    -- Calculate total sales for each year
    SUM(sales_amount) AS [Total Sales],

    -- Calculate the cumulative total sales across years
    SUM(SUM(sales_amount)) OVER (
        ORDER BY YEAR(order_date)
    ) AS [Cumulative Total Sales]

FROM fact_sales

-- Group sales by year
GROUP BY YEAR(order_date)

-- Display the results in chronological order
ORDER BY YEAR(order_date);
