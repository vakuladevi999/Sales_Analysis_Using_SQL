-- Year-over-Year (YoY) Sales Analysis
-- Compare each year's sales with the previous year's sales
-- to identify the sales difference and growth direction.

SELECT
    [Order Year],
    [Current Year Sales],
    [Previous Year Sales],

    -- Calculate the difference between current and previous year sales
    [Current Year Sales] - [Previous Year Sales] AS [Sales Difference],

    -- Identify whether sales increased, decreased, or remained unchanged
    CASE
        WHEN [Current Year Sales] > [Previous Year Sales] THEN 'Increase'
        WHEN [Current Year Sales] < [Previous Year Sales] THEN 'Decrease'
        ELSE 'No Change'
    END AS [Sales Trend]

FROM (
    SELECT
        YEAR(order_date) AS [Order Year],

        -- Calculate total sales for each year
        SUM(sales_amount) AS [Current Year Sales],

        -- Retrieve the previous year's total sales
        LAG(SUM(sales_amount)) OVER (
            ORDER BY YEAR(order_date)
        ) AS [Previous Year Sales]

    FROM fact_sales

    -- Group sales by year
    GROUP BY YEAR(order_date)
) AS F;
